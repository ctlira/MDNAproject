# ---------------------------------------------------------
# GammaPA — Shared library (template)
# ---------------------------------------------------------

# --- Explicit ordered source lists (order matters) ----------------------------

# GammaPA source files
set(GAMMAPA_SOURCES_LCL
    ../SourceCodeUA/0ModuleUAData.f90
    ../SourceCodeUA/UaDataTools.f90
    ../Main_GammaPA/GammaPaMain.f90
    ../Main_GammaPA/gamma_pa.f90)

# ---------------------------------------------------------
# Export source list to parent scope for fortls
# ---------------------------------------------------------
set(GAMMAPA_SOURCES
    ${GAMMAPA_SOURCES_LCL}
    PARENT_SCOPE
)

# --- Target -------------------------------------------------------------------

add_executable(GammaPA ${GAMMAPA_SOURCES_LCL})

# --- Debugger working directory ----------------------------------------------

# This forces Visual Studio to run from the root repository directory
set_property(TARGET GammaPA PROPERTY
    DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
)

# --- Intel Fortran flags (same as Main_PGLdll) --------------------------------

if(CMAKE_Fortran_COMPILER_ID MATCHES "Intel|IntelLLVM")
    # For Release/RelWithDebInfo, force the standalone executable to bake in the runtime
    target_compile_options(GammaPA PRIVATE
        $<$<NOT:$<CONFIG:Debug>>:/libs:static>
    )
endif()

# --- Fixed-form handling for .for and .f files --------------------------------

foreach(src ${SRC_GPA})
    if(src MATCHES "\\.(for|f)$")
        set_source_files_properties(${src} PROPERTIES COMPILE_OPTIONS "/fixed")
    endif()
endforeach()
